⚙️ গুগল অ্যাপস স্ক্রিপ্ট কোড (Full CRUD Apps Script):

গুগল শিটের Extensions → Apps Script এ গিয়ে নিচের পূর্ণাঙ্গ কোডটি পেস্ট করে Deploy > New Deployment > Web app (Who has access: Anyone) হিসেবে সেভ করুন:

function doGet(e) {
  var sheetName = (e.parameter.sheet || 'messages').trim();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    // case-insensitive fallback
    var all = ss.getSheets();
    for (var i = 0; i < all.length; i++) {
      if (all[i].getName().toLowerCase() === sheetName.toLowerCase()) {
        sheet = all[i]; break;
      }
    }
  }
  if (!sheet) {
    return ContentService.createTextOutput(JSON.stringify({error: "Sheet not found: " + sheetName}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) {
    return ContentService.createTextOutput(JSON.stringify([]))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var headers = data[0];
  var rows = [];
  for (var r = 1; r < data.length; r++) {
    var obj = { id: 'row-' + r };
    for (var c = 0; c < headers.length; c++) {
      obj[headers[c]] = data[r][c];
    }
    rows.push(obj);
  }
  return ContentService.createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var postData = JSON.parse(e.postData.contents);
    var sheetName = (postData.sheet || e.parameter.sheet || 'messages').trim();
    var action = postData.action || 'insert'; // insert, update, delete
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
    }
    var headers = sheet.getRange(1, 1, 1, Math.max(1, sheet.getLastColumn())).getValues()[0];
    
    if (action === 'insert') {
      var row = [];
      for (var i = 0; i < headers.length; i++) {
        var col = headers[i];
        row.push(postData.data[col] !== undefined ? postData.data[col] : '');
      }
      sheet.appendRow(row);
      return ContentService.createTextOutput(JSON.stringify({success: true, message: "Row added"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    // Update support
    if (action === 'update') {
      var updateRowIndex = parseInt(postData.rowIndex, 10);
      if (updateRowIndex > 1 && updateRowIndex <= sheet.getLastRow()) {
        for (var i = 0; i < headers.length; i++) {
          var col = headers[i];
          if (postData.data[col] !== undefined) {
            sheet.getRange(updateRowIndex, i + 1).setValue(postData.data[col]);
          }
        }
        return ContentService.createTextOutput(JSON.stringify({success: true, message: "Row updated"}))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    // Delete support
    if (action === 'delete') {
      var rowIndex = parseInt(postData.rowIndex, 10);
      if (rowIndex > 1 && rowIndex <= sheet.getLastRow()) {
        sheet.deleteRow(rowIndex);
        return ContentService.createTextOutput(JSON.stringify({success: true, message: "Row deleted"}))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({success: true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
